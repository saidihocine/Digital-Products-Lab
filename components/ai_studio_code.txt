'use client';

import { useState, type FormEvent } from 'react';
import { Check, Mail } from 'lucide-react';

const EMAIL_PATTERN = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
const STORAGE_KEY = 'digital_products_lab_subscribers';

/**
 * مكون اشتراك بريدي بسيط وهادئ في أسفل الصفحة الرئيسية
 * لجمع البريد الإلكتروني لإشعارات القوالب الرقمية القادمة.
 */
export function EmailSubscription() {
  const [email, setEmail] = useState<string>('');
  const [status, setStatus] = useState<'idle' | 'submitted' | 'error'>('idle');
  const [errorMessage, setErrorMessage] = useState<string>('');

  const handleSubmit = (event: FormEvent<HTMLFormElement>): void => {
    event.preventDefault();
    const trimmed = email.trim().toLowerCase();

    if (!EMAIL_PATTERN.test(trimmed)) {
      setStatus('error');
      setErrorMessage('يرجى إدخال بريد إلكتروني صحيح.');
      return;
    }

    try {
      const existingRaw = window.localStorage.getItem(STORAGE_KEY);
      const existing: string[] = existingRaw ? (JSON.parse(existingRaw) as string[]) : [];
      if (!existing.includes(trimmed)) {
        window.localStorage.setItem(STORAGE_KEY, JSON.stringify([...existing, trimmed]));
      }
    } catch {
      // الاكتفاء بحالة التأكيد المرئية في حال تعذر التخزين المحلي
    }

    setErrorMessage('');
    setStatus('submitted');
    setEmail('');
  };

  return (
    <section
      aria-labelledby="newsletter-heading"
      className="border-t border-[#E5E2D9] bg-[#F8F7F4]"
    >
      <div className="mx-auto max-w-[1200px] px-6 py-20 lg:py-24">
        <div className="mx-auto max-w-2xl rounded-2xl border border-[#E5E2D9] bg-white p-8 text-center shadow-[0_2px_14px_-6px_rgba(28,28,26,0.05)] sm:p-12">
          <div className="mx-auto flex h-10 w-10 items-center justify-center rounded-lg border border-[#ECEAE4] bg-[#F8F7F4] text-[#575650]">
            <Mail className="h-4 w-4 stroke-[1.6]" />
          </div>

          <h2
            id="newsletter-heading"
            className="mt-5 text-2xl font-semibold tracking-tight text-[#1C1C1A] sm:text-3xl"
          >
            إشعارات الإصدارات الجديدة
          </h2>

          <p className="mx-auto mt-3 max-w-md text-sm leading-[1.85] text-[#575650]">
            اترك بريدك الإلكتروني ليصلك تنبيه هادئ عند نشر قوالب A4 جديدة أو تحديثات في مختبر Digital-Products-Lab. دون رسائل مزعجة.
          </p>

          {status === 'submitted' ? (
            <div
              role="status"
              className="mx-auto mt-8 flex max-w-md items-center justify-center gap-2.5 rounded-lg border border-[#D8D5CD] bg-[#F8F7F4] px-5 py-3.5 text-sm font-medium text-[#1C1C1A]"
            >
              <Check className="h-4 w-4 shrink-0 text-[#1C1C1A]" />
              <span>تم تسجيل بريدك بنجاح. سنراسلك عند توفر قوالب جديدة.</span>
            </div>
          ) : (
            <form
              onSubmit={handleSubmit}
              noValidate
              className="mx-auto mt-8 flex max-w-md flex-col gap-3 sm:flex-row"
            >
              <label htmlFor="newsletter-email" className="sr-only">
                البريد الإلكتروني
              </label>
              <input
                id="newsletter-email"
                type="email"
                inputMode="email"
                autoComplete="email"
                dir="ltr"
                value={email}
                onChange={(e) => {
                  setEmail(e.target.value);
                  if (status === 'error') {
                    setStatus('idle');
                    setErrorMessage('');
                  }
                }}
                placeholder="name@example.com"
                aria-invalid={status === 'error'}
                className="min-h-[46px] flex-1 rounded-lg border border-[#DCD8CE] bg-[#FAF9F6] px-4 py-2.5 text-left text-sm text-[#1C1C1A] placeholder:text-[#9E9B92] transition-colors focus:border-[#1C1C1A] focus:bg-white focus:outline-none"
              />
              <button
                type="submit"
                className="inline-flex min-h-[46px] shrink-0 items-center justify-center whitespace-nowrap rounded-lg border border-[#1C1C1A] bg-[#1C1C1A] px-6 py-2.5 text-sm font-medium text-[#FAF9F6] transition-colors duration-200 hover:bg-[#2F2F2B] focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-[#1C1C1A]"
              >
                اشترك
              </button>
            </form>
          )}

          {status === 'error' && errorMessage && (
            <p role="alert" className="mt-3 text-xs font-medium text-[#8A3B2B]">
              {errorMessage}
            </p>
          )}
        </div>
      </div>
    </section>
  );
}
